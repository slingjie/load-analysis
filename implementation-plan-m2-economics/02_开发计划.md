## 🔧 M1: 数据层扩展 (2h)

### 任务 1.1: 更新 `types.ts` (1h)

**文件**: `types.ts`

**新增接口**:
```typescript
// 经济性参数接口
export interface EconomicsParams {
  capexPerWh: number;                    // 单位投资成本（元/Wh）
  projectYears: number;                  // 项目年限（年）
  omCostRatio: number;                   // 年运维成本占 CAPEX 比例
  firstYearDecayRate: number;            // 首年衰减率
  subsequentDecayRate: number;           // 后续年衰减率
  cellReplacementYear?: number;          // 电芯更换年份（可选）
  cellReplacementCostRatio?: number;     // 更换成本比例（可选）
}

// 经济性结果接口
export interface StorageEconomicsResult {
  staticPaybackYears: number;
  irr: number;
  npv: number;
  annual_cash_flows: {
    year: number;
    revenue: number;
    opex: number;
    cash_flow: number;
    cumulative: number;
  }[];
}

// 扩展 BatchCapacityItem（在现有基础上添加）
export interface BatchCapacityItem {
  capacityKwh: number;
  yearEqCycles: number;
  firstYearProfit: number;
  response?: any;
  status: 'computing' | 'done' | 'error';
  errorMsg?: string;
}

// 全局分析状态
export interface StorageAnalysisState {
  batchResults: BatchCapacityItem[];
  storageParams: StorageParamsTemplate;
  selectedCapacityForEconomics?: {
    capacityKwh: number;
    firstYearProfit: number;
    yearEqCycles: number;
  };
  economicsParams: EconomicsParams;
  economicsResults: Map<number, StorageEconomicsResult>;
  comparisonBasket: Set<number>;
}
```

**操作**:
- [ ] 在 `types.ts` 底部添加上述接口定义
- [ ] TypeScript 编译检查通过
