### 3️⃣ hooks/useStorageAnalysis.ts — 全局状态 Hook (新建)

**位置**: `src/hooks/useStorageAnalysis.ts`

**代码框架**:
```typescript
import { useState, useCallback } from 'react';
import type {
  StorageAnalysisState,
  BatchCapacityItem,
  EconomicsParams,
  StorageEconomicsResult,
} from '@/types';
import { ECONOMICS_PARAMS_DEFAULT } from '@/constants';

/**
 * 全局储能分析状态管理 Hook
 */
export const useStorageAnalysis = () => {
  const [state, setState] = useState<StorageAnalysisState>({
    batchResults: [],
    storageParams: {},
    economicsParams: ECONOMICS_PARAMS_DEFAULT,
    economicsResults: new Map(),
    comparisonBasket: new Set(),
  });

  // ========== 1. 选择容量进行经济性分析 ==========
  const selectCapacityForEconomics = useCallback(
    (item: BatchCapacityItem) => {
      setState((prev) => ({
        ...prev,
        selectedCapacityForEconomics: {
          capacityKwh: item.capacityKwh,
          firstYearProfit: item.firstYearProfit,
          yearEqCycles: item.yearEqCycles,
        },
      }));
    },
    []
  );

  // ========== 2. 更新经济性参数 ==========
  const updateEconomicsParams = useCallback(
    (params: Partial<EconomicsParams>) => {
      setState((prev) => ({
        ...prev,
        economicsParams: { ...prev.economicsParams, ...params },
      }));
    },
    []
  );

  // ========== 3. 保存经济性结果 ==========
  const saveEconomicsResult = useCallback(
    (capacityKwh: number, result: StorageEconomicsResult) => {
      setState((prev) => {
        const newResults = new Map(prev.economicsResults);
        newResults.set(capacityKwh, result);
        return { ...prev, economicsResults: newResults };
      });
    },
    []
  );

  // ========== 4. 切换对比篮 ==========
  const toggleComparison = useCallback((capacityKwh: number) => {
    setState((prev) => {
      const newBasket = new Set(prev.comparisonBasket);
      if (newBasket.has(capacityKwh)) {
        newBasket.delete(capacityKwh);
      } else if (newBasket.size < 5) {
        newBasket.add(capacityKwh);
      }
      return { ...prev, comparisonBasket: newBasket };
    });
  }, []);

  // ========== 5. 清空对比篮 ==========
  const clearBasket = useCallback(() => {
    setState((prev) => ({
      ...prev,
      comparisonBasket: new Set(),
    }));
  }, []);

  // ========== 6. 更新批量结果 (可选) ==========
  const setBatchResults = useCallback((results: BatchCapacityItem[]) => {
    setState((prev) => ({
      ...prev,
      batchResults: results,
    }));
  }, []);

  return {
    state,
    selectCapacityForEconomics,
    updateEconomicsParams,
    saveEconomicsResult,
    toggleComparison,
    clearBasket,
    setBatchResults,
  };
};

export default useStorageAnalysis;
```

```typescript
export const StorageCyclesPage = () => {
  const { state, toggleComparison, clearBasket, selectCapacityForEconomics } =
    useStorageAnalysis();

  const [isComputing, setIsComputing] = useState(false);
  const [progress, setProgress] = useState(0);
  const [currentCapacity, setCurrentCapacity] = useState<number | null>(null);

  const [capacityConfig, setCapacityConfig] = useState<CapacityConfig>({
    startKwh: 5000,
    stepKwh: 500,
    stepCount: 10,
  });

  // 批量计算逻辑（并发度 3）
  const handleBatchCompute = async () => {
    setIsComputing(true);
    const results: BatchCapacityItem[] = [];
    const capacities = Array.from(
      { length: capacityConfig.stepCount },
      (_, i) => capacityConfig.startKwh + i * capacityConfig.stepKwh
    );

    const concurrency = 3;
    for (let i = 0; i < capacities.length; i += concurrency) {
      const batch = capacities.slice(i, i + concurrency);
      await Promise.all(
        batch.map(async (capacity) => {
          setCurrentCapacity(capacity);
          try {
            const response = await computeStorageCycles(
              capacity,
              state.storageParams
            );
            results.push({
              capacityKwh: capacity,
              yearEqCycles: response.year_eq_cycles,
              firstYearProfit: response.profit_net,
              response,
              status: 'done',
            });
          } catch (error) {
            results.push({
              capacityKwh: capacity,
              yearEqCycles: 0,
              firstYearProfit: 0,
              response: null,
              status: 'error',
              errorMsg: (error as Error).message,
            });
          }
          setProgress(Math.round(((i + batch.indexOf(capacity) + 1) / capacities.length) * 100));
        })
      );
    }

    // 更新全局状态
    // setState(prev => ({ ...prev, batchResults: results }));

    setIsComputing(false);
    setProgress(0);
    setCurrentCapacity(null);
  };

  return (
    <div className="storage-cycles-page">
      <h1>💾 批量容量对比</h1>

      {/* 配置区 */}
      <section className="batch-config">
        <h3>容量范围配置</h3>
        <div className="form-group">
          <label>起始容量 (kWh)</label>
          <input
            type="number"
            value={capacityConfig.startKwh}
            onChange={(e) =>
              setCapacityConfig({
                ...capacityConfig,
                startKwh: parseInt(e.target.value),
              })
            }
          />
        </div>
        {/* 其他输入字段... */}
        <button
          onClick={handleBatchCompute}
          disabled={isComputing}
          className="btn btn-primary"
        >
          {isComputing ? '计算中...' : '🚀 开始批量容量对比'}
        </button>
      </section>

      {/* 进度条 */}
      {isComputing && (
        <div className="progress-section">
          {/* 进度条实现... */}
        </div>
      )}

      {/* 结果展示 */}
      {state.batchResults.length > 0 && (
        <>
          <BatchCapacityTable
            data={state.batchResults}
            comparisonBasket={state.comparisonBasket}
            onToggleComparison={toggleComparison}
            onViewEconomics={selectCapacityForEconomics}
          />
          <BatchCapacityChart data={state.batchResults} />
        </>
      )}
    </div>
  );
};

export default StorageCyclesPage;
```

**代码框架**:
```typescript
import { useEffect, useRef } from 'react';
import Chart from 'chart.js/auto';
import type { StorageEconomicsResult } from '@/types';
import './CashFlowChart.css';

interface Props {
  data: StorageEconomicsResult['annual_cash_flows'];
}

export const CashFlowChart = ({ data }: Props) => {
  const chartRef = useRef<HTMLCanvasElement>(null);
  const chartInstance = useRef<Chart | null>(null);

  useEffect(() => {
    if (!chartRef.current || data.length === 0) return;

    // 销毁旧图表
    if (chartInstance.current) {
      chartInstance.current.destroy();
    }

    // 创建新图表
    chartInstance.current = new Chart(chartRef.current, {
      type: 'bar',
      data: {
        labels: data.map((d) => `Y${d.year}`),
        datasets: [
          {
            label: '年度现金流 (元)',
            data: data.map((d) => d.cash_flow),
            backgroundColor: data.map((d) =>
              d.cash_flow >= 0 ? '#10b981' : '#ef4444'
            ),
            yAxisID: 'y',
          },
          {
            label: '累计现金流 (元)',
            data: data.map((d) => d.cumulative),
            borderColor: '#3b82f6',
            type: 'line',
            yAxisID: 'y1',
            tension: 0.4,
          },
        ],
      },
      options: {
        responsive: true,
        interaction: { mode: 'index', intersect: false },
        plugins: {
          legend: { display: true, position: 'top' },
        },
        scales: {
          y: {
            type: 'linear',
            display: true,
            position: 'left',
            title: { display: true, text: '现金流 (元)' },
          },
          y1: {
            type: 'linear',
            display: true,
            position: 'right',
            title: { display: true, text: '累计现金流 (元)' },
          },
        },
      },
    });

    return () => {
      if (chartInstance.current) {
        chartInstance.current.destroy();
      }
    };
  }, [data]);

  return (
    <div className="cash-flow-chart-container">
      <canvas ref={chartRef} height="80"></canvas>
    </div>
  );
};

export default CashFlowChart;
```

### 9️⃣ ComparisonModal.tsx (新建)

**位置**: `src/components/ComparisonModal.tsx`

**代码框架**:
```typescript
import { useState } from 'react';
import type { BatchCapacityItem, StorageEconomicsResult } from '@/types';
import ComparisonChart from './ComparisonChart';
import './ComparisonModal.css';

interface Props {
  items: BatchCapacityItem[];
  economicsResults: Map<number, StorageEconomicsResult>;
  onClose: () => void;
}

export const ComparisonModal = ({ items, economicsResults, onClose }: Props) => {
  const [showEconomics, setShowEconomics] = useState(false);

  const sortedItems = [...items].sort((a, b) => a.capacityKwh - b.capacityKwh);

  return (
    <div className="modal-overlay" onClick={onClose}>
      <div className="modal-content" onClick={(e) => e.stopPropagation()}>
        <div className="modal-header">
          <h2>📊 容量对比 ({items.length} 个)</h2>
          <button className="close-btn" onClick={onClose}>
            ✕
          </button>
        </div>

        <div className="modal-body">
          {/* 对比图表 */}
          <ComparisonChart
            items={sortedItems}
            economicsResults={economicsResults}
            showEconomics={showEconomics}
          />

          {/* 对比表格 */}
          <table className="comparison-table">
            <thead>
              <tr>
                <th>容量 (kWh)</th>
                <th>循环数 (次/年)</th>
                <th>收益 (万元)</th>
                {showEconomics && (
                  <>
                    <th>回收期 (年)</th>
                    <th>IRR (%)</th>
                  </>
                )}
              </tr>
            </thead>
            <tbody>
              {sortedItems.map((item) => {
                const economics = economicsResults.get(item.capacityKwh);
                return (
                  <tr key={item.capacityKwh}>
                    <td>
                      <strong>{item.capacityKwh.toLocaleString()}</strong>
                    </td>
                    <td>{item.yearEqCycles.toFixed(0)}</td>
                    <td>{(item.firstYearProfit / 10000).toFixed(1)}</td>
                    {showEconomics && (
                      <>
                        <td>
                          {economics?.staticPaybackYears.toFixed(1) || '-'}
                        </td>
                        <td>
                          {economics
                            ? (economics.irr * 100).toFixed(1)
                            : '-'}
                        </td>
                      </>
                    )}
                  </tr>
                );
              })}
            </tbody>
          </table>
        </div>

        <div className="modal-footer">
          <button
            onClick={() => setShowEconomics(!showEconomics)}
            className="btn btn-secondary"
          >
            {showEconomics ? '隐藏' : '显示'} 经济性数据
          </button>
          <button onClick={onClose} className="btn btn-primary">
            关闭
          </button>
        </div>
      </div>
    </div>
  );
};

export default ComparisonModal;
```
